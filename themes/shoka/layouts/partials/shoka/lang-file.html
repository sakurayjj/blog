{{- $lang := .lang | default "en" -}} {{- $parts := split $lang "-" -}} {{- if gt (len $parts) 1 -}} {{- $lang = printf "%s-%s" (index $parts 0) (upper (index $parts 1)) -}} {{- end -}} {{- $themeDirs := slice -}} {{- if fileExists "themes" -}} {{- range readDir "themes" -}} {{- if .IsDir -}} {{- $themeDirs = $themeDirs | append .Name -}} {{- end -}} {{- end -}} {{- end -}} {{- $path := printf "languages/%s.yml" $lang -}} {{- if not (fileExists $path) -}} {{- $path = "" -}} {{- range $themeDirs -}} {{- $candidate := printf "themes/%s/languages/%s.yml" . $lang -}} {{- if and (eq $path "") (fileExists $candidate) -}} {{- $path = $candidate -}} {{- end -}} {{- end -}} {{- end -}} {{- if not $path -}} {{- $defaultPath := "languages/default.yml" -}} {{- if not (fileExists $defaultPath) -}} {{- $defaultPath = "" -}} {{- range $themeDirs -}} {{- $candidate := printf "themes/%s/languages/default.yml" . -}} {{- if and (eq $defaultPath "") (fileExists $candidate) -}} {{- $defaultPath = $candidate -}} {{- end -}} {{- end -}} {{- end -}} {{- if $defaultPath -}} {{- $defaultFile := trim (readFile $defaultPath) " \r\n\t" -}} {{- if $defaultFile -}} {{- $path = printf "languages/%s" $defaultFile -}} {{- if not (fileExists $path) -}} {{- $path = "" -}} {{- range $themeDirs -}} {{- $candidate := printf "themes/%s/languages/%s" . $defaultFile -}} {{- if and (eq $path "") (fileExists $candidate) -}} {{- $path = $candidate -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if not $path -}} {{- $path = "languages/en.yml" -}} {{- if not (fileExists $path) -}} {{- $path = "" -}} {{- range $themeDirs -}} {{- $candidate := printf "themes/%s/languages/en.yml" . -}} {{- if and (eq $path "") (fileExists $candidate) -}} {{- $path = $candidate -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $path -}}
